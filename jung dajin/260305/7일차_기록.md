#include <jni.h>
#include <string>
#include <vector>
#include <chrono> // 시간 측정용
#include <torch/script.h>

extern "C" JNIEXPORT jfloatArray JNICALL
Java_com_example_shutter_1android_1ai_MainActivity_detectDeepfake(
        JNIEnv* env,
        jobject /* this */,
        jstring modelPath,
        jfloatArray audioData) {

    // 결과 저장을 위한 배열 (0: score, 1: load_time, 2: inference_time)
    jfloatArray resultMap = env->NewFloatArray(3);
    float results[3] = { -1.0f, 0.0f, 0.0f };

    const char* path = env->GetStringUTFChars(modelPath, nullptr);

    // 1. 모델 로드 시간 측정 시작
    auto start_load = std::chrono::high_resolution_clock::now();

    torch::jit::script::Module module;
    try {
        module = torch::jit::load(path);
    } catch (const c10::Error& e) {
        env->ReleaseStringUTFChars(modelPath, path);
        results[0] = -1.0f; // 로드 실패
        env->SetFloatArrayRegion(resultMap, 0, 3, results);
        return resultMap;
    }

    auto end_load = std::chrono::high_resolution_clock::now();
    results[1] = std::chrono::duration<float, std::milli>(end_load - start_load).count();

    // 2. 데이터 준비
    jfloat* data = env->GetFloatArrayElements(audioData, nullptr);
    jsize length = env->GetArrayLength(audioData);
    auto options = torch::TensorOptions().dtype(torch::kFloat32);
    torch::Tensor input_tensor = torch::from_blob(data, {1, length}, options);

    // 3. 추론 시간 측정 시작
    auto start_inference = std::chrono::high_resolution_clock::now();

    std::vector<torch::jit::IValue> inputs;
    inputs.push_back(input_tensor);
    at::Tensor output = module.forward(inputs).toTensor();
    float score = output.item<float>();

    auto end_inference = std::chrono::high_resolution_clock::now();
    results[2] = std::chrono::duration<float, std::milli>(end_inference - start_inference).count();
    results[0] = score;

    // 4. 메모리 해제 및 결과 반환
    env->ReleaseFloatArrayElements(audioData, data, JNI_ABORT);
    env->ReleaseStringUTFChars(modelPath, path);

    env->SetFloatArrayRegion(resultMap, 0, 3, results);
    return resultMap;
}

package com.example.shutter_android_ai

import android.content.Context
import android.media.MediaCodec
import android.media.MediaExtractor
import android.media.MediaFormat
import android.os.Bundle
import android.widget.Button
import android.widget.TextView
import androidx.appcompat.app.AppCompatActivity
import com.example.shutter_android_ai.R
import java.io.File
import java.io.FileOutputStream
import java.nio.ByteOrder
import kotlin.concurrent.thread

class MainActivity : AppCompatActivity() {

    // C++ JNI 함수 선언
    private external fun detectDeepfake(modelPath: String, audioData: FloatArray): FloatArray

    companion object {
        init {
            System.loadLibrary("opencv_java4")
            System.loadLibrary("shutter_android_ai")
        }
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        val btnTestInference = findViewById<Button>(R.id.btnTestInference)
        val tvResult = findViewById<TextView>(R.id.tvResult)

        btnTestInference.setOnClickListener {
            tvResult.text = "분석 중..."

            thread {
                try {
                    val modelPath = getAssetFilePath(this@MainActivity, "voice_deepfake_model.pt")
                    val audioPath = getAssetFilePath(this@MainActivity, "test.flac")
                    val audioData = decodeAudioFromFile(audioPath, 64000)

                    // 2. 결과를 배열로 받습니다.
                    val results: FloatArray = detectDeepfake(modelPath, audioData)

                    // 3. 배열 인덱스를 통해 안전하게 값을 가져옵니다.
                    val score = results[0]
                    val loadTime = results[1]
                    val inferenceTime = results[2]

                    runOnUiThread {
                        if (score <= -1.0f) { // -1.0 이하면 로드 실패로 간주
                            tvResult.text = "에러: 모델 로드 실패"
                        } else {
                            val resultText = """
                                [추론 성공]
                                Score: ${String.format("%.4f", score)}
                                Load: ${String.format("%.2f", loadTime)} ms
                                Infer: ${String.format("%.2f", inferenceTime)} ms
                            """.trimIndent()
                            tvResult.text = resultText
                        }
                    }
                } catch (e: Exception) {
                    runOnUiThread {
                        tvResult.text = "에러 발생: ${e.message}"
                    }
                }
            }
        }
    }

    // 수정된 디코딩 함수: Context 대신 로컬 파일 경로(filePath)를 직접 받습니다.
    private fun decodeAudioFromFile(filePath: String, targetLength: Int): FloatArray {
        val floatList = mutableListOf<Float>()
        val extractor = MediaExtractor()
        
        // assets.openFd 대신 일반 파일 경로를 사용하여 소스를 설정합니다. (압축 문제 해결)
        extractor.setDataSource(filePath)

        var format: MediaFormat? = null
        var mime: String? = null
        for (i in 0 until extractor.trackCount) {
            val trackFormat = extractor.getTrackFormat(i)
            val trackMime = trackFormat.getString(MediaFormat.KEY_MIME)
            if (trackMime?.startsWith("audio/") == true) {
                format = trackFormat
                mime = trackMime
                extractor.selectTrack(i)
                break
            }
        }

        if (format == null || mime == null) throw IllegalArgumentException("오디오 트랙을 찾을 수 없습니다.")

        val codec = MediaCodec.createDecoderByType(mime)
        codec.configure(format, null, null, 0)
        codec.start()

        val info = MediaCodec.BufferInfo()
        var isEOS = false

        while (true) {
            if (!isEOS) {
                val inIndex = codec.dequeueInputBuffer(10000)
                if (inIndex >= 0) {
                    val buffer = codec.getInputBuffer(inIndex)
                    val sampleSize = extractor.readSampleData(buffer!!, 0)
                    if (sampleSize < 0) {
                        codec.queueInputBuffer(inIndex, 0, 0, 0, MediaCodec.BUFFER_FLAG_END_OF_STREAM)
                        isEOS = true
                    } else {
                        codec.queueInputBuffer(inIndex, 0, sampleSize, extractor.sampleTime, 0)
                        extractor.advance()
                    }
                }
            }

            val outIndex = codec.dequeueOutputBuffer(info, 10000)
            if (outIndex >= 0) {
                val buffer = codec.getOutputBuffer(outIndex)
                if (buffer != null && info.size > 0) {
                    buffer.position(info.offset)
                    buffer.limit(info.offset + info.size)
                    buffer.order(ByteOrder.LITTLE_ENDIAN)
                    val shortBuffer = buffer.asShortBuffer()
                    
                    while (shortBuffer.hasRemaining()) {
                        floatList.add(shortBuffer.get().toFloat() / 32768.0f)
                    }
                }
                codec.releaseOutputBuffer(outIndex, false)
            } else if (outIndex == MediaCodec.INFO_TRY_AGAIN_LATER) {
                if (isEOS) break
            }
            if ((info.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM) != 0) break
        }

        codec.stop()
        codec.release()
        extractor.release()

        val result = FloatArray(targetLength)
        val copyLength = minOf(floatList.size, targetLength)
        for (i in 0 until copyLength) {
            result[i] = floatList[i]
        }
        return result
    }

    private fun getAssetFilePath(context: Context, assetName: String): String {
        val file = File(context.filesDir, assetName)
        if (file.exists() && file.length() > 0) return file.absolutePath

        context.assets.open(assetName).use { inputStream ->
            FileOutputStream(file).use { outputStream ->
                inputStream.copyTo(outputStream)
            }
        }
        return file.absolutePath
    }
}

## 카카오톡/sns 기반 영상 통화 중 음성 패킷 취득 가능 여부 조사
요약: 카카오톡, 텔레그램 등 대부분의 서드파티 메신저 앱은 기본적으로 `allowAudioPlaybackCapture="true"` 상태이거나 시스템 접근을 허용하고 있어 기술적으로 캡처가 가능하다. 반면, 기본 전화 앱은 통신비밀보호법과 구글의 강력한 보안 정책으로 인해 시스템 차원에서 캡처가 원천 차단되어 있다.


### 1. 왜 메신저 앱은 되고, 기본 전화는 안 되는거지?

안드로이드 OS는 오디오 데이터를 '속성(Usage)'에 따라 다르게 취급한다.

기본 전화 (PSTN): 통화 속성이 `USAGE_VOICE_COMMUNICATION_SIGNALLING` 등으로 분류되며, 이는 프라이버시 보호를 위해 어떤 앱도 접근할 수 없도록 OS 커널 수준에서 막혀 있다. (제조사 서명키가 필요한 이유)
메신저 앱 (VoIP): 카카오톡 등의 영상/음성 통화는 기술적으로 '데이터 통신을 통한 미디어 재생'에 가깝다. 안드로이드 10부터 도입된 정책에 따라, 앱 개발자가 명시적으로 거부(`false`) 설정을 하지 않는 한 다른 앱이 오디오를 캡처할 수 있도록 허용하는 것이 기본값이다. (Default가 true)


### 2. 주요 앱별 캡처 가능 여부 (조사 결과)

| 앱 구분 | 캡처 가능 여부 | 이유 및 특징 |
| --- | --- | --- |
| 카카오톡 | 가능 | `allowAudioPlaybackCapture`를 따로 막지 않아 `MediaProjection`을 통한 루프백 캡처가 가능하다. |
| 텔레그램 | 가능 | 오픈소스 기반이며 보안을 중시하지만, 시스템 오디오 공유 기능을 위해 캡처를 허용하고 있다. |
| 디스코드 | 가능 | 화면 공유 기능이 핵심인 앱이라 오디오 캡처에 가장 개방적이다. |
| 기본 전화 | 불가능 | `AudioAttributes.ALLOW_CAPTURE_BY_NONE` 수준의 보호를 받으며, 법적 규제로 인해 접근이 차단된다. |
| 유튜브/넷플릭스 | 불가능/제한 | 저작권 보호(DRM)가 걸린 콘텐츠는 캡처 시 소리가 0(무음)으로 들어온다. |


### 3. 기술적 세부사항

안드로이드 공식 문서에 따르면, 앱의 `targetSdkVersion`에 따라 기본 동작이 달라진다.

Android 10 (API 29) 이상을 타겟팅하는 앱: 별 설정을 안 하면 기본이 `true`다. 카카오톡과 텔레그램은 최신 안드로이드 버전을 준수하므로 별도의 차단 로직을 넣지 않는 한 캡처 대상이 된다.
시스템 제약 사항: 설령 `true`이더라도, 상대방 앱이 '음성 통화(USAGE_VOICE_COMMUNICATION)' 속성을 정확히 사용하고 있어야 한다. 만약 카카오톡이 오디오 속성을 특수하게 처리한다면 캡처가 안 될 수도 있지만, 현재까지는 표준 VoIP 방식을 따르고 있어 캡처가 원활하다.


### 4. 개발 시 팁: 캡처가 안 될 때의 체크리스트

만약 특정 메신저 앱의 소리가 캡처되지 않는다면 다음 두 가지를 확인해야 한다.

1. 캡처 앱의 설정: 내 앱의 `AudioRecord` 설정에서 `addMatchingUsage()`를 통해 `USAGE_VOICE_COMMUNICATION`과 `USAGE_MEDIA`를 모두 추가했는지 확인하기.
2. 시스템 상태: 현재 기기에서 다른 앱(예: 화면 녹화 앱)이 이미 오디오를 선점하고 있다면 캡처가 거부될 수 있다.
